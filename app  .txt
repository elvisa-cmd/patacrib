class SavedSearch(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    user_id = db.Column(db.Integer, db.ForeignKey('user.id'), nullable=False)
    city = db.Column(db.String(100))
    estate = db.Column(db.String(100))
    min_price = db.Column(db.Float)
    max_price = db.Column(db.Float)
    bedrooms = db.Column(db.Integer)
    created_at = db.Column(db.DateTime, default=datetime.utcnow)
    last_notified = db.Column(db.DateTime)
    
    @property
    def url(self):
        url = '/search?'
        if self.city:
            url += f'city={self.city}&'
        if self.estate:
            url += f'q={self.estate}&'
        if self.bedrooms:
            url += f'bedrooms={self.bedrooms}&'
        if self.min_price:
            url += f'min_price={self.min_price}&'
        if self.max_price:
            url += f'max_price={self.max_price}&'
        return url
    
    @property
    def new_count(self):
        query = Property.query.filter_by(status='available')
        if self.city:
            query = query.filter_by(city=self.city)
        if self.estate:
            query = query.filter(Property.estate.contains(self.estate))
        if self.bedrooms:
            query = query.filter(Property.bedrooms >= self.bedrooms)
        if self.min_price:
            query = query.filter(Property.price >= self.min_price)
        if self.max_price:
            query = query.filter(Property.price <= self.max_price)
        if self.last_notified:
            query = query.filter(Property.created_at > self.last_notified)
        return query.count()

@app.template_filter('from_json')
def from_json_filter(value):
    if value:
        try:
            return json.loads(value)
        except:
            return []
    return []

@app.template_filter('format_ksh')
def format_ksh_filter(amount):
    try:
        if amount >= 1000000:
            return f"KSh {amount/1000000:.1f}M"
        elif amount >= 1000:
            return f"KSh {amount/1000:.0f}K"
        else:
            return f"KSh {amount:,.0f}"
    except:
        return f"KSh {amount}"

def allowed_file(filename):
    return '.' in filename and filename.rsplit('.', 1)[1].lower() in app.config['ALLOWED_EXTENSIONS']

def save_image(file, subfolder='properties'):
    try:
        filename = secure_filename(file.filename)
        ext = filename.rsplit('.', 1)[1].lower()
        new_filename = f"{uuid.uuid4().hex}.{ext}"
        filepath = os.path.join(app.config['UPLOAD_FOLDER'], subfolder, new_filename)
        os.makedirs(os.path.dirname(filepath), exist_ok=True)
        file.save(filepath)
        return f'/static/uploads/{subfolder}/{new_filename}'
    except:
        return None

def save_base64_image(base64_str, subfolder='webcam'):
    try:
        if 'base64,' in base64_str:
            base64_str = base64_str.split('base64,')[1]
        img_data = base64.b64decode(base64_str)
        img = Image.open(io.BytesIO(img_data))
        filename = f"{uuid.uuid4().hex}.jpg"
        filepath = os.path.join(app.config['UPLOAD_FOLDER'], subfolder, filename)
        os.makedirs(os.path.dirname(filepath), exist_ok=True)
        img.save(filepath, 'JPEG', quality=85)
        return f'/static/uploads/{subfolder}/{filename}'
    except:
        return None

def admin_required(f):
    @wraps(f)
    def decorated_function(*args, **kwargs):
        if not current_user.is_authenticated or current_user.user_type != 'admin':
            flash('Admin access required', 'error')
            return redirect(url_for('index'))
        return f(*args, **kwargs)
    return decorated_function

def get_neighborhood_safety(city, estate):
    safety = NeighborhoodSafety.query.filter_by(city=city, estate=estate).first()
    if safety:
        return {
            'score': safety.safety_score,
            'police_distance': safety.police_distance,
            'street_lights': safety.street_lights,
            'night_safety': safety.night_safety,
            'reviews': safety.total_reviews,
            'safe_percentage': safety.safe_percentage
        }
    return {
        'score': 75,
        'police_distance': 1.5,
        'street_lights': True,
        'night_safety': 'Moderate',
        'reviews': 18,
        'safe_percentage': 78
    }

def get_water_schedule(city, estate):
    water = WaterSchedule.query.filter_by(city=city, estate=estate).first()
    if water:
        return {
            'supplier': water.supplier,
            'today': water.schedule_today,
            'tomorrow': water.schedule_tomorrow
        }
    return {
        'supplier': 'Nairobi Water',
        'today': '8am - 12pm',
        'tomorrow': '2pm - 6pm'
    }

def get_matatu_routes(estate):
    routes = MatatuRoute.query.filter(
        (MatatuRoute.from_area.contains(estate)) | 
        (MatatuRoute.to_area.contains(estate)) |
        (MatatuRoute.via.contains(estate))
    ).limit(3).all()
    
    if routes:
        return [{'number': r.route_number, 'sacco': r.sacco} for r in routes]
    
    return [{'number': '44', 'sacco': 'KBS'}, {'number': '105', 'sacco': 'Eastlands'}, {'number': '33C', 'sacco': 'Ngong Rd'}]

@app.route('/property/<int:property_id>')
def property_detail(property_id):
    property = Property.query.get_or_404(property_id)
    
    if current_user.is_authenticated:
        view = PropertyView(property_id=property_id, user_id=current_user.id)
        db.session.add(view)
        db.session.commit()
        
        if property.owner:
            property.owner.last_active = datetime.utcnow()
            db.session.commit()
    
    images = json.loads(property.images) if property.images else []
    tour_photos = json.loads(property.tour_photos) if property.tour_photos else []
    features = json.loads(property.features) if property.features else []
    amenities = json.loads(property.amenities) if property.amenities else []
    
    avg_price = db.session.query(db.func.avg(Property.price)).filter(
        Property.city == property.city,
        Property.property_type == property.property_type,
        Property.status == 'available',
        Property.id != property_id
    ).scalar() or property.price
    
    safety_data = get_neighborhood_safety(property.city, property.estate)
    water_data = get_water_schedule(property.city, property.estate)
    matatu_routes = get_matatu_routes(property.estate)
    stage_distance = property.stage_distance or 300
    cbd_time = property.cbd_travel_time or 25
    
    is_saved = False
    if current_user.is_authenticated:
        saved = SavedProperty.query.filter_by(user_id=current_user.id, property_id=property_id).first()
        is_saved = saved is not None
    
    return render_template('property_detail.html',
                         property=property,
                         images=images,
                         tour_photos=tour_photos,
                         features=features,
                         amenities=amenities,
                         is_saved=is_saved,
                         avg_price=avg_price,
                         safety=safety_data,
                         water=water_data,
                         matatu_routes=matatu_routes,
                         stage_distance=stage_distance,
                         cbd_time=cbd_time)

@app.route('/search')
def search():
    query = request.args.get('q', '')
    city = request.args.get('city', '')
    property_type = request.args.get('type', '')
    
    properties = Property.query.filter_by(status='available')
    
    if query:
        properties = properties.filter(
            db.or_(
                Property.title.contains(query),
                Property.description.contains(query),
                Property.address.contains(query),
                Property.estate.contains(query)
            )
        )
    if city:
        properties = properties.filter_by(city=city)
    if property_type:
        properties = properties.filter_by(property_type=property_type)
    
    properties = properties.order_by(Property.created_at.desc()).all()
    cities = db.session.query(Property.city).distinct().all()
    cities = [c[0] for c in cities if c[0]]
    
    return render_template('search.html',
                         properties=properties,
                         cities=cities,
                         query=query,
                         city=city,
                         property_type=property_type)

@app.route('/login', methods=['GET', 'POST'])
def login():
    if current_user.is_authenticated:
        return redirect(url_for('dashboard'))
    
    if request.method == 'POST':
        email = request.form.get('email')
        password = request.form.get('password')
        
        user = User.query.filter_by(email=email).first()
        
        if user and check_password_hash(user.password, password):
            login_user(user)
            flash(f'Welcome back, {user.name}!', 'success')
            return redirect(url_for('dashboard'))
        else:
            flash('Invalid email or password', 'error')
    
    return render_template('login.html')

@app.route('/signup', methods=['GET', 'POST'])
def signup():
    if current_user.is_authenticated:
        return redirect(url_for('index'))
    
    if request.method == 'POST':
        email = request.form.get('email')
        password = request.form.get('password')
        name = request.form.get('name')
        phone = request.form.get('phone')
        user_type = request.form.get('user_type')
        
        if not email or not password or not name or not phone or not user_type:
            flash('All fields are required', 'error')
            return redirect(url_for('signup'))
        
        if User.query.filter_by(email=email).first():
            flash('Email already registered', 'error')
            return redirect(url_for('signup'))
        
        user = User(
            email=email,
            password=generate_password_hash(password),
            name=name,
            phone=phone,
            user_type=user_type
        )
        
        db.session.add(user)
        db.session.commit()
        
        login_user(user)
        flash('Account created successfully!', 'success')
        return redirect(url_for('dashboard'))
    
    return render_template('signup.html')

@app.route('/property/add', methods=['GET', 'POST'])
@login_required
@admin_required
def add_property():
    if request.method == 'POST':
        try:
            title = request.form.get('title')
            description = request.form.get('description')
            price = float(request.form.get('price', 0))
            price_type = request.form.get('price_type', 'month')
            bedrooms = int(request.form.get('bedrooms', 0))
            bathrooms = int(request.form.get('bathrooms', 0))
            property_type = request.form.get('property_type')
            
            address = request.form.get('address', '')
            city = request.form.get('city', '')
            estate = request.form.get('estate')
            
            gps_latitude = request.form.get('latitude')
            gps_longitude = request.form.get('longitude')
            gps_accuracy = request.form.get('gps_accuracy')
            
            if not gps_latitude or not gps_longitude:
                flash('❌ You must capture your exact GPS location first!', 'error')
                return redirect(url_for('add_property'))
            
            latitude = float(gps_latitude)
            longitude = float(gps_longitude)
            accuracy = float(gps_accuracy) if gps_accuracy else None
            
            safety_data = get_neighborhood_safety(city, estate)
            water_data = get_water_schedule(city, estate)
            matatu_routes = get_matatu_routes(estate)
            
            images = []
            
            if 'images' in request.files:
                files = request.files.getlist('images')
                for file in files:
                    if file and allowed_file(file.filename):
                        path = save_image(file)
                        if path:
                            images.append(path)
            
            webcam_images = request.form.getlist('webcam_images[]')
            for img_data in webcam_images:
                if img_data:
                    path = save_base64_image(img_data)
                    if path:
                        images.append(path)
            
            video = None
            if 'video' in request.files:
                video_file = request.files['video']
                if video_file and allowed_file(video_file.filename):
                    video = save_video(video_file)
            
            if 'recorded_video' in request.form and request.form['recorded_video']:
                recorded_video_url = request.form['recorded_video']
                if recorded_video_url.startswith('/static/uploads/videos/'):
                    video = recorded_video_url
            
            # NEW: Handle 360° tour photos
            tour_photos = []
            if 'tour_photos' in request.files:
                files = request.files.getlist('tour_photos')
                for file in files:

@app.route('/property/edit/<int:property_id>', methods=['GET', 'POST'])
@login_required
@admin_required
def edit_property(property_id):
    property = Property.query.get_or_404(property_id)
    
    if property.admin_id != current_user.id:
        flash('Unauthorized', 'error')
        return redirect(url_for('index'))
    
    if request.method == 'POST':
        try:
            property.title = request.form.get('title')
            property.description = request.form.get('description')
            property.price = float(request.form.get('price', 0))
            property.price_type = request.form.get('price_type')
            property.bedrooms = int(request.form.get('bedrooms', 0))
            property.bathrooms = int(request.form.get('bathrooms', 0))
            property.property_type = request.form.get('property_type')
            property.status = request.form.get('status', 'available')
            
            property.address = request.form.get('address', '')
            property.city = request.form.get('city', '')
            property.estate = request.form.get('estate')
            
            gps_latitude = request.form.get('latitude')
            gps_longitude = request.form.get('longitude')
            gps_accuracy = request.form.get('gps_accuracy')
            
            if gps_latitude and gps_longitude and gps_latitude != '' and gps_longitude != '':
                property.latitude = float(gps_latitude)
                property.longitude = float(gps_longitude)
                property.gps_accuracy = float(gps_accuracy) if gps_accuracy else None
            
            current_images = json.loads(property.images) if property.images else []
            
            if 'images' in request.files:
                files = request.files.getlist('images')
                for file in files:
                    if file and allowed_file(file.filename):
                        path = save_image(file)
                        if path:
                            current_images.append(path)
            
            webcam_images = request.form.getlist('webcam_images[]')
            for img_data in webcam_images:
                if img_data:
                    path = save_base64_image(img_data)
                    if path:
                        current_images.append(path)
            
            property.images = json.dumps(current_images)
            
            # NEW: Handle 360° tour photos for edit
            current_tour_photos = json.loads(property.tour_photos) if property.tour_photos else []
            if 'tour_photos' in request.files:
                files = request.files.getlist('tour_photos')
                for file in files:
                    if file and allowed_file(file.filename):
                        path = save_image(file, 'tours')
                        if path:
                            current_tour_photos.append(path)
            property.tour_photos = json.dumps(current_tour_photos)
            
            if 'video' in request.files:
                video_file = request.files['video']
                if video_file and allowed_file(video_file.filename):
                    property.video = save_video(video_file)
            
            property.features = json.dumps(request.form.getlist('features[]'))
            property.amenities = json.dumps(request.form.getlist('amenities[]'))
            
            db.session.commit()
            flash('Property updated successfully!', 'success')
            return redirect(url_for('property_detail', property_id=property.id))
        except Exception as e:
            flash(f'Error updating property: {str(e)}', 'error')
    
    images = json.loads(property.images) if property.images else []
    tour_photos = json.loads(property.tour_photos) if property.tour_photos else []
    features = json.loads(property.features) if property.features else []
    amenities = json.loads(property.amenities) if property.amenities else []
    
    return render_template('edit_property.html',
                         property=property,
                         images=images,
                         tour_photos=tour_photos,
                         features=features,
                         amenities=amenities)
